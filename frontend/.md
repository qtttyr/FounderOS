const GLASSES_IP = () => localStorage.getItem('glasses_ip')
export async function ping() {
 const res = await fetch(`http://${GLASSES_IP()}/ping`)
 return res.ok
}
export async function sendCmd(cmd) {
 await fetch(`http://${GLASSES_IP()}/cmd`, {
 method: 'POST',
 headers: { 'Content-Type': 'application/json' },
 body: JSON.stringify(cmd)
 })
}
export async function updateHUD({ time, date, temp, city, projects, tasks }) {
 await sendCmd({ cmd: 'update', time, date, temp, city, projects, tasks })
}
export async function sendAI(text) {
 await sendCmd({ cmd: 'ai', text })
}
export async function sendNotify(text) {
 await sendCmd({ cmd: 'notify', text })
}
